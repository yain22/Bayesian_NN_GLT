.bnn_class_log_probs <- function(logits, outcome) {
  if (outcome == "binary") {
    eta <- logits[, 1L]
    return(cbind(stats::plogis(-eta, log.p = TRUE), stats::plogis(eta, log.p = TRUE)))
  }
  all_logits <- cbind(logits, 0)
  largest <- cbind(seq_len(nrow(logits)), max.col(all_logits, ties.method = "first"))
  shifted <- all_logits - all_logits[largest]
  # Separate the leading exp(0) so a nearly certain class retains its small
  # negative log probability even when 1 + sum(other terms) rounds to one.
  other_terms <- exp(shifted)
  other_terms[largest] <- 0
  shifted - log1p(rowSums(other_terms))
}

.bnn_class_score <- function(log_probs, y, outcome) {
  if (outcome == "binary") {
    # Evaluate the small complementary probability directly near a perfect fit.
    return(matrix(ifelse(y == 2L, exp(log_probs[, 1L]), -exp(log_probs[, 2L])), ncol = 1L))
  }
  last <- ncol(log_probs)
  score <- -exp(log_probs[, -last, drop = FALSE])
  rows <- which(y != last)
  observed <- cbind(rows, y[rows])
  score[observed] <- -expm1(log_probs[observed])
  score
}

.gltbnn_engine_class_draws <- function(fit, x, draws, summarize = FALSE) {
  shape <- c(nrow(x), fit$config$n_classes)
  result <- array(0, dim = if (summarize) shape else c(shape, draws))
  for (b in seq_len(draws)) {
    weights <- .bnn_draw_weights(fit$parameters, .bnn_normal_draw(fit$parameters))
    logits <- .bnn_forward(x, weights)$logits
    if (any(!is.finite(logits))) {
      stop("Predicted logits exceed the finite numerical range. Check newdata and its predictor units.",
           call. = FALSE)
    }
    probabilities <- exp(.bnn_class_log_probs(logits, fit$config$outcome))
    if (summarize) {
      result <- result + probabilities
    } else {
      result[, , b] <- probabilities
    }
  }
  if (summarize) result / draws else result
}

.bnn_scalar_locals <- function(A, B, d = 1L, normalizer_only = FALSE) {
  if (length(A) != length(B) || any(!is.finite(A) | A <= 0 | !is.finite(B) | B <= 0)) {
    stop("Scalar local kernel requires one positive finite A and B for each coefficient.")
  }
  log_a <- log(A)
  log_b <- log(B)
  m <- (log_b - log_a) / 3
  log_q <- (2 * log_a + log_b) / 3
  q <- exp(log_q)
  # The endpoints suppress the exponential tails by at least exp(-70) on
  # ordinary scales, including the first-moment integrand.
  upper <- log1p(80 / q + 14 / sqrt(q))
  lower <- -.5 * log1p(160 / q + 28 / sqrt(q))
  tiny <- q < 1e-300
  if (any(tiny)) {
    upper[tiny] <- log(80 + 14 * sqrt(q[tiny]) + q[tiny]) - log_q[tiny]
    lower[tiny] <- -.5 * (log(160 + 28 * sqrt(q[tiny]) + q[tiny]) - log_q[tiny])
  }
  if (any(!is.finite(upper) | !is.finite(lower))) stop("Local scale exceeds quadrature range.")
  segments <- pmax(1L, ceiling(pmax(-lower, upper) / 6))
  z <- raw_first <- raw_log <- raw_energy <- numeric(length(q))
  for (count in unique(segments)) {
    idx <- which(segments == count)
    Q <- q[idx]
    tiny_q <- tiny[idx]
    total <- first <- log_scale <- energy <- numeric(length(idx))
    for (width in list(lower[idx] / count, upper[idx] / count)) {
      for (panel in seq_len(count)) {
        for (j in seq_along(.bnn_scalar_rule$nodes)) {
          s <- width * ((panel - 1L) + .bnn_scalar_rule$nodes[j])
          cost <- Q * (expm1(s) + .5 * expm1(-2 * s))
          if (any(tiny_q)) {
            b <- log_q[idx[tiny_q]]
            u <- s[tiny_q]
            cost[tiny_q] <- exp(b + u) + .5 * exp(b - 2*u) - 1.5*Q[tiny_q]
          }
          small <- abs(s) < 1e-4
          if (any(small)) {
            a <- s[small]
            cost[small] <- (Q[small] * a) * a * (1.5 + a * (-.5 + a * (.375 - .125 * a)))
          }
          if (d != 1L) cost <- cost + (d - 1L) * s
          weight <- abs(width) * .bnn_scalar_rule$weights[j]
          mass <- weight * exp(-cost)
          total <- total + mass
          first_exponent <- s - cost
          if (any(tiny_q)) first_exponent[tiny_q] <- first_exponent[tiny_q] + log_q[idx[tiny_q]]
          first <- first + weight * exp(first_exponent)
          log_scale <- log_scale + mass * s
          energy <- energy + mass * cost
        }
      }
    }
    z[idx] <- total
    raw_first[idx] <- first
    raw_log[idx] <- log_scale
    raw_energy[idx] <- energy
  }
  # The d=2 grouped first moment needs only the d=1 log normalizer. Its
  # unused scalar moments can exceed double precision for subnormal rates.
  if (normalizer_only) return(log(z) - 1.5*q + (1-d)*m)
  zeta <- m + raw_log / z
  log_u <- m + log(raw_first / z)
  if (any(tiny)) log_u[tiny] <- log_u[tiny] - log_q[tiny]
  # Integration by parts yields B E[lambda^-2] = A E[lambda] exactly.
  u <- exp(log_u)
  log_phi <- log_a - log_b + log_u
  if (d != 1L) log_phi <- log_phi + log1p((d-1L)*exp(-log_a-log_u))
  phi <- exp(log_phi)
  entropy <- log(z) + zeta + raw_energy / z
  if (any(!is.finite(u) | u <= 0 | !is.finite(phi) | phi <= 0 | !is.finite(entropy))) {
    stop("Nonfinite scalar local moments.")
  }
  list(u = u, phi = phi, zeta = zeta, entropy = entropy, A = A, B = B,
       log_normalizer = log(z) - 1.5 * q + (1-d)*m)
}

    }
  }

.bnn_coefficient_terms <- function(g, prior) {
  M <- g$M
  if (prior == "GLT") {
    C <- M * g$tau$zeta + sum(g$nu$zeta + g$nu$phi)
    hyper <- -M * g$tau$zeta - sum(g$nu$zeta) - g$tau$phi * sum(g$local$u * g$nu$phi) +
      sum(g$local$entropy) + g$tail$log_normalizer - g$tail$log_prior_support -
      (C - g$tail$C) * g$tail$r - 2 * g$tau$zeta - sum(g$nu$zeta) - g$tau$phi +
      g$tau$entropy + sum(g$nu$entropy)
    return(list(precision = g$local$phi, logvar = 2 * g$local$zeta, hyper = hyper))
  }
  hyper <- .bnn_ig_log_prior(g$tau, .5, g$aux$phi, -g$aux$zeta) + g$tau$entropy +
    .bnn_ig_log_prior(g$aux, .5, M^2, log(M^2)) + g$aux$entropy
  if (prior == "G") {
    return(list(precision = rep(g$tau$phi, M), logvar = rep(g$tau$zeta, M), hyper = hyper))
  }
  hyper <- hyper + sum(.bnn_ig_log_prior(g$local, .5, g$nu$phi, -g$nu$zeta) +
    g$local$entropy + .bnn_ig_log_prior(g$nu, .5, 1, 0) + g$nu$entropy)
  list(precision = g$tau$phi * g$local$phi, logvar = g$tau$zeta + g$local$zeta, hyper = hyper)
}

.bnn_coefficient_update <- function(g, prior, moments, phi) {
  M <- g$M
  if (length(moments) != M) stop("Coefficient moments must cover each weight and bias.")
  if (prior == "GLT") {
    g$local <- .bnn_scalar_locals(g$tau$phi * g$nu$phi, phi * moments)
    g$tau <- .bnn_ig(1 + M * (1 + g$tail$r), 1 + sum(g$local$u * g$nu$phi))
    g$nu <- .bnn_local_ig(1 + g$tail$r, g$local$u * g$tau$phi + g$tail$r)
    g$tail <- .bnn_coefficient_tail(M, M * g$tau$zeta + sum(g$nu$zeta + g$nu$phi), g$fixed_shape)
  } else {
    local_precision <- 1
    if (prior == "GL") {
      g$local <- .bnn_local_ig(1, g$nu$phi + phi * moments * g$tau$phi / 2)
      local_precision <- g$local$phi
      g$nu <- .bnn_local_ig(1, 1 + local_precision)
    }
    g$tau <- .bnn_ig((M + 1) / 2, g$aux$phi + phi * sum(moments * local_precision) / 2)
    g$aux <- .bnn_ig(1, M^2 + g$tau$phi)
  }
  g
}

.bnn_weight_objective <- function(parameters, x, y, phi, rates, epsilon, gradient = FALSE,
                                  outcome = "continuous") {
  moments <- .bnn_weight_moments(parameters, coefficient = TRUE)
  value <- sum(vapply(parameters$logsd, sum, numeric(1))) -
    .5 * sum(vapply(seq_along(moments), function(l) sum(rates[[l]] * moments[[l]]), numeric(1)))
  if (gradient) gm <- gs <- lapply(parameters$mu, function(m) m * 0)
  draws <- length(epsilon)
  for (z in epsilon) {
    weights <- .bnn_draw_weights(parameters, z)
    forward <- .bnn_forward(x, weights)
    if (outcome == "continuous") {
      residual <- y - forward$y
      value <- value - .5 * phi * sum(residual^2) / draws
      if (gradient) score <- matrix(phi * residual, ncol = 1L)
    } else {
      log_probs <- .bnn_class_log_probs(forward$logits, outcome)
      value <- value + phi * .bnn_class_loglik(log_probs, y) / draws
      if (gradient) score <- phi * .bnn_class_score(log_probs, y, outcome)
    }
    if (gradient) {
      back <- .bnn_backprop(score, forward, weights)
      for (l in seq_along(weights)) {
        grad <- back[[l]] / draws
        gm[[l]] <- gm[[l]] + grad
        gs[[l]] <- gs[[l]] + grad * exp(parameters$logsd[[l]]) * z[[l]]
      }
    }
  }
  if (!gradient) return(value)
  for (l in seq_along(gm)) {
    gm[[l]] <- gm[[l]] - rates[[l]] * parameters$mu[[l]]
    gs[[l]] <- gs[[l]] + 1 - rates[[l]] * exp(2 * parameters$logsd[[l]])
  }
  list(value = value, gradient = c(gm, gs))
}

.bnn_elbo <- function(state, parameters, x, y, priors, epsilon, noise_prior = NULL,
                      outcome = "continuous") {
  moments <- .bnn_weight_moments(parameters, coefficient = TRUE)
  output <- length(moments)
  gaussian_entropy <- function(ls) sum((1 + log(2 * pi)) / 2 + ls)
  output_dim <- length(parameters$mu[[output]])
  hidden_log_scale_sq <- 0
  hidden_precision <- 1
  noise_terms <- NULL
  if (outcome == "continuous") {
    noise <- state$noise
    hidden_log_scale_sq <- noise$zeta
    hidden_precision <- noise$phi
    sse <- .bnn_sse(parameters, x, y, epsilon)
    observation <- -length(y) * (log(2 * pi) + noise$zeta) / 2 - noise$phi * mean(sse) / 2
    centered_draws <- noise$phi * (mean(sse) - sse) / 2
    noise_terms <- if (is.null(noise_prior)) {
      c(noise_prior = -noise$zeta, noise_entropy = noise$entropy)
    } else {
      c(noise_neg_kl = -.bnn_ig_kl(noise, noise_prior[["shape"]], noise_prior[["scale"]]))
    }
  } else {
    loglik <- vapply(epsilon, function(z) {
      weights <- .bnn_draw_weights(parameters, z)
      logits <- .bnn_forward(x, weights)$logits
      .bnn_class_loglik(.bnn_class_log_probs(logits, outcome), y)
    }, numeric(1))
    observation <- mean(loglik)
    centered_draws <- loglik - observation
  }
  terms <- c(
    observation = observation,
    noise_terms,
    output_weight_prior = -output_dim * (log(2 * pi) + state$u$zeta) / 2 -
      state$u$phi * sum(moments[[output]]) / 2,
    output_weight_entropy = gaussian_entropy(parameters$logsd[[output]]),
    output_u_prior = .bnn_ig_log_prior(state$u, .5, state$rho$phi, -state$rho$zeta),
    output_rho_prior = .bnn_ig_log_prior(state$rho, .5, 1 / 25, log(1 / 25)),
    output_hyper_entropy = state$u$entropy + state$rho$entropy,
    hidden_weight_prior = 0, hidden_entropy = 0, prior_hyper = 0
  )
  for (l in seq_along(priors)) {
    pr <- priors[[l]]
    terms["hidden_weight_prior"] <- terms["hidden_weight_prior"] - .5 * sum(
      log(2 * pi) + hidden_log_scale_sq + pr$logvar + hidden_precision * moments[[l]] * pr$precision)
    terms["hidden_entropy"] <- terms["hidden_entropy"] + gaussian_entropy(parameters$logsd[[l]])
    terms["prior_hyper"] <- terms["prior_hyper"] + pr$hyper
  }
  if (any(!is.finite(terms))) stop("Nonfinite ELBO component during fitting.")
  value <- sum(terms)
  pairs <- .bnn_pair_means(centered_draws)
  summary <- if (outcome == "continuous") list(expected_sse = mean(sse)) else list(expected_loglik = observation)
  c(list(value = value, components = terms), summary,
    list(mcse = stats::sd(pairs) / sqrt(length(pairs)), sample_values = value + centered_draws))
}

.bnn_adam_step <- function(parameters, adam, iteration, x, y, hidden_precision, rates, control,
                           outcome = "continuous", n_total = nrow(x)) {
  epsilon <- .bnn_antithetic_draws(parameters, control$weight_draws)
  likelihood_precision <- hidden_precision * (n_total / nrow(x))
  objective <- .bnn_weight_objective(parameters, x, y, likelihood_precision, rates, epsilon, TRUE, outcome)
  grad <- lapply(objective$gradient, function(g) g / n_total)
  norm <- sqrt(sum(vapply(grad, function(g) sum(g^2), numeric(1))))
  if (!is.finite(objective$value)) stop("Nonfinite weight objective.")
  if (!is.finite(norm)) {
    # Finite gradient entries can overflow when squared. The unit direction
    # and the clipping decision both remain available on a rescaled vector.
    largest <- max(vapply(grad, function(g) max(abs(g)), numeric(1)))
    if (!is.finite(largest)) stop("Nonfinite weight gradient.")
    scaled <- lapply(grad, function(g) g / largest)
    scaled_norm <- sqrt(sum(vapply(scaled, function(g) sum(g^2), numeric(1))))
    norm <- largest * scaled_norm
    if (norm > control$grad_clip) {
      grad <- lapply(scaled, function(g) (g / scaled_norm) * control$grad_clip)
    }
  } else if (norm > control$grad_clip) {
    grad <- lapply(grad, function(g) g * control$grad_clip / norm)
  }
  adam$m <- Map(function(m, g) .9 * m + .1 * g, adam$m, grad)
  adam$v <- Map(function(v, g) .999 * v + .001 * g^2, adam$v, grad)
  denominator <- lapply(adam$v, function(v) sqrt(v / (1 - .999^iteration)) + 1e-8)
  direction <- Map(function(m, den) (m / (1 - .9^iteration)) / den, adam$m, denominator)
  slopes <- Map(function(g, d) sum(g * d), grad, direction)
  fallback <- sum(unlist(slopes)) <= 0
  if (fallback) direction <- Map(`/`, grad, denominator)

  # Reuse the gradient draws for candidate scoring so that sampling noise
  # cannot produce spurious improvements within this backtracking step.
  accepted <- FALSE
  gain <- 0
  indices <- seq_along(parameters$mu)
  for (bt in 0:control$max_backtracks) {
    scale <- control$lr * 2^(-bt)
    candidate <- Map(function(p, d) p + scale * d, c(parameters$mu, parameters$logsd), direction)
    cp <- list(mu = candidate[indices], logsd = candidate[length(indices) + indices])
    cp$logsd <- lapply(cp$logsd, function(z) {
      z[] <- pmax(control$logsd_bounds[1], pmin(control$logsd_bounds[2], as.vector(z)))
      z
    })
    value <- .bnn_weight_objective(cp, x, y, likelihood_precision, rates, epsilon, outcome = outcome)
    if (is.finite(value) && value >= objective$value) {
      parameters <- cp
      gain <- value - objective$value
      accepted <- TRUE
      break
    }
  }
  list(parameters = parameters, adam = adam,
       record = data.frame(step = iteration, accepted = accepted, backtracks = bt,
                           direction_fallback = fallback, sample_objective = objective$value,
                           sample_gain = gain, gradient_norm = norm))
}

.gltbnn_engine_fit <- function(x, y, hidden, prior, control, seed, noise_prior = NULL,
                               outcome = "continuous", n_classes = NULL, geometry = "coefficient",
                               initial_weights = NULL, progress = NULL, fixed_shape = NULL,
                               batches = NULL, milestones = integer(), on_milestone = NULL) {

    if (is.null(initial_weights)) {
      warm_batches <- .bnn_batch_schedule(nrow(x), control$batch_size, control$init_steps, stream(1000))
      initial_weights <- .bnn_initialize(x, y, hidden, seed, control$init_steps, outcome,
                                         n_classes, batches = warm_batches)

                                   as.double(control$outer) * control$weight_steps, stream(3000))

      !is.finite(fixed_shape) || fixed_shape <= .5)) stop("A fixed tail requires coefficient GLT and shape > 1/2.")

  group_sizes <- if (coefficient) rep(1L, length(hidden)) else dimensions
  weights <- initial_weights
  if (is.null(weights)) weights <- .bnn_initialize(x, y, hidden, seed, control$init_steps, outcome, n_classes)

                u = .bnn_ig(2, 2), rho = .bnn_ig(1, 1.04))

    hidden_precision <- state$noise$phi
  }
  layer_priors <- function(groups) Map(function(g, k) .bnn_prior_terms(g, k, prior), groups, group_counts)
  priors <- layer_priors(state$g)

.bnn_backprop <- function(back, forward, weights) {
  gradient <- vector("list", length(weights))
  for (layer in rev(seq_along(weights))) {
    gradient[[layer]] <- crossprod(back, forward$features[[layer]])
    if (layer > 1L) {
      w <- weights[[layer]]
      back <- (back %*% w[, -ncol(w), drop = FALSE]) * (forward$pre[[layer - 1L]] > 0)
    }
  }
  gradient

.bnn_initialize <- function(x, y, hidden, seed, steps, outcome = "continuous", n_classes = NULL,
                            batches = NULL) {
  set.seed(seed)
  inputs <- c(ncol(x), hidden) + 1L
  output_units <- if (outcome == "categorical") n_classes - 1L else 1L
  units <- c(hidden, output_units)
  weights <- lapply(seq_along(units), function(l) {
    matrix(stats::rnorm(units[l] * inputs[l], sd = sqrt(2 / inputs[l])), nrow = units[l])
  })
  first <- second <- lapply(weights, function(w) w * 0)
  for (iteration in seq_len(steps)) {
    if (is.null(batches)) {
      batch_x <- x
      batch_y <- y
    } else {
      ids <- batches[[iteration]]
      batch_x <- x[ids, , drop = FALSE]
      batch_y <- y[ids]
    }
    forward <- .bnn_forward(batch_x, weights)
    if (outcome == "continuous") {
      back <- matrix((forward$y - batch_y) / length(batch_y), ncol = 1L)
    } else if (outcome == "binary" && !is.null(batches)) {
      back <- (stats::plogis(forward$logits) - (batch_y - 1L)) / length(batch_y)
    } else {
      log_probs <- .bnn_class_log_probs(forward$logits, outcome)
      back <- -.bnn_class_score(log_probs, batch_y, outcome) / length(batch_y)
    }
    gradient <- .bnn_backprop(back, forward, weights)
    for (l in seq_along(weights)) {
      first[[l]] <- .9 * first[[l]] + .1 * gradient[[l]]
      second[[l]] <- .999 * second[[l]] + .001 * gradient[[l]]^2
      weights[[l]] <- weights[[l]] - .003 * (first[[l]] / (1 - .9^iteration)) /
        (sqrt(second[[l]] / (1 - .999^iteration)) + 1e-8)
    }
  }
  weights
}

.bnn_ig <- function(a, b) {
  if (!is.finite(a) || !is.finite(b) || a <= 0 || b <= 0) {
    stop("Invalid inverse-gamma variational parameters.")
  }
  if (a >= 1e4) {
    # Stirling expansions avoid subtracting terms of order a * log(a).
    inv <- 1 / a
    psi_gap <- inv / 2 + inv^2 / 12 - inv^4 / 120
    remainder <- inv / 12 - inv^3 / 360 + inv^5 / 1260
    log_ratio <- .bnn_log_ratio(b, a)
    zeta <- log_ratio + psi_gap
    entropy <- log_ratio + .5 * (log(2 * pi) - log(a)) + remainder + (a * psi_gap + psi_gap)
  } else {
    zeta <- log(b) - digamma(a)
    entropy <- a + log(b) + lgamma(a) - (a + 1) * digamma(a)
  }
  list(a = a, b = b, phi = a / b, zeta = zeta, entropy = entropy)
}

.bnn_product_parts <- function(x, y) {
  splitter <- 2^27 + 1
  cx <- splitter * x
  cy <- splitter * y
  x_high <- cx - (cx - x)
  y_high <- cy - (cy - y)
  x_low <- x - x_high
  y_low <- y - y_high
  product <- x * y
  error <- (((x_high * y_high - product) + x_high * y_low) + x_low * y_high) + x_low * y_low
  c(product, error)
}

.bnn_log_precision_ratio <- function(q, shape, scale) {
  # log{(q$a / q$b) / (shape / scale)}. Scaling by powers of two keeps
  # the products finite without rounding a division of the input parameters.
  values <- c(q$a, scale, shape, q$b)
  powers <- pmin(floor(log2(values)), .Machine$double.max.exp - 1L)
  m <- values / 2^powers
  shift <- (powers[1L] + powers[2L]) - (powers[3L] + powers[4L])
  if (abs(shift) > 2) return(shift * log(2) + .bnn_log_ratio(m[1L] * m[2L], m[3L] * m[4L]))
  numerator <- .bnn_product_parts(m[1L] * 2^shift, m[2L])
  denominator <- .bnn_product_parts(m[3L], m[4L])
  difference <- (numerator[1L] - denominator[1L]) + (numerator[2L] - denominator[2L])
  log1p(difference / denominator[1L])
}

.bnn_ig_kl <- function(q, shape, scale) {
  s <- .bnn_log_ratio(shape, q$a)
  shape_loss <- 0
  if (s != 0) {
    # Integrate the Taylor remainder of lgamma(t) - t*log(t) + t over
    # log shape. s and the exponential weight are scaled out near equality
    # and at large positive s, respectively.
    integrand <- function(v) {
      weight <- if (s > 1) exp((v - 1) * s) * (-expm1(-v * s)) else expm1(v * s) / s
      weight * .bnn_gamma_curvature(log(shape) - v * s)
    }
    integral <- stats::integrate(integrand, 0, 1, rel.tol = 1e-10, abs.tol = 0)$value
    shape_loss <- if (s > 1) exp(s + log(s * integral)) else s^2 * integral
  }
  z <- .bnn_log_precision_ratio(q, shape, scale)
  # shape * (exp(z) - 1 - z). Squaring after rescaling preserves small
  # divergences; the factored positive-tail form avoids overflow in exp(z).
  if (abs(z) < 1e-3) {
    mean_loss <- (sqrt(shape) * z)^2 * (.5 + z * (1/6 + z * (1/24 + z * (1/120 + z/720))))
  } else if (z > 1) {
    mean_loss <- exp(log(shape) + z + log1p(-(1 + z) * exp(-z)))
  } else {
    mean_loss <- shape * (expm1(z) - z)
  }
  kl <- shape_loss + mean_loss
  if (!is.finite(kl)) stop("The inverse-gamma noise KL exceeds the finite numerical range.")
  kl
}

.bnn_quad_local_integral <- function(A, B, d, power, rel_tol, subdivisions) {
  coefficient <- 1 - d + power
  la <- log(A); lb <- log(B)
  derivative <- function(t) {
    z <- max(if (coefficient == 0) -Inf else log(abs(coefficient)), la + t, lb - 2 * t)
    term <- if (coefficient == 0) 0 else sign(coefficient) * exp(log(abs(coefficient)) - z)
    term - exp(la + t - z) + exp(lb - 2 * t - z)
  }
  midpoint <- (lb - la) / 3
  radius <- 1
  for (j in 1:60) {
    lower <- midpoint - radius; upper <- midpoint + radius
    if (derivative(lower) > 0 && derivative(upper) < 0) break
    radius <- radius * 2
  }
  if (!(derivative(lower) > 0 && derivative(upper) < 0))
    stop("Could not bracket a local quadrature mode.")
  mode <- stats::uniroot(derivative, c(lower, upper), tol = 1e-12)$root

  integral <- .bnn_quad_integrate(density, -Inf, Inf, rel_tol, subdivisions)

  list(log_value = log_height + log(scale) + log(integral$value),
       mode = mode, scale = scale, density = density, offset = offset, integral = integral)

.bnn_local_factor <- function(A, B, d, rel_tol = 1e-10,
                                     subdivisions = 2000L) {
  .bnn_quad_quadrature_settings(rel_tol, subdivisions)
  .bnn_quad_positive(c(A, B), "Local kernel parameters")
  if (length(A) != 1L || length(B) != 1L || length(d) != 1L ||
      !is.finite(d) || d < 1 || d != as.integer(d)) stop("Invalid local group dimension.")
  log_q <- (2 * log(A) + log(B)) / 3
  broad <- log_q < log(1e-3)
  if ((broad && d == 1L) || log_q > log(1e4) + 2 * log(d)) {
    # The scalar rule resolves a broad d=1 plateau. In the concentrated
    # regime, the (1-d)s tilt shifts the mode by less than one hundredth of
    # a kernel SD; direct moments preserve the entropy on the lambda scale.
    ans <- .bnn_scalar_locals(A, B, d)
    ans$d <- d
    ans$identity_residual <- B * ans$phi - A * ans$u - (d - 1)
    ans$estimated_relative_integration_error <- NA_real_
    return(ans)
  }
  base <- .bnn_quad_local_integral(A, B, d, 0, rel_tol, subdivisions)
  first <- if (d == 2L && broad) {
    # The numerator of E[lambda] has the d=1 kernel. Its broad log-scale
    # plateau is resolved by the scalar rule's split panels.
    list(log_value = .bnn_scalar_locals(A,B,normalizer_only=TRUE))
  } else {
    .bnn_quad_local_integral(A, B, d, 1, rel_tol, subdivisions)
  }
  inverse <- .bnn_quad_local_integral(A, B, d, -2, rel_tol, subdivisions)
  centered <- .bnn_quad_integrate(function(z) z * base$density(z),
    -Inf, Inf, rel_tol, subdivisions)
  # Express moment ratios relative to the denominator mode before adding
  # their large log heights. This avoids subtracting nearly equal heights.
  log_ratio <- function(numerator, power) {
    if (is.null(numerator$integral)) return(numerator$log_value - base$log_value)
    base$offset(numerator$mode - base$mode) + power*numerator$mode +
      log(numerator$scale / base$scale) + log(numerator$integral$value / base$integral$value)
  }
  u <- exp(log_ratio(first,1))
  phi <- exp(log_ratio(inverse,-2))
  zeta <- base$mode + base$scale * centered$value / base$integral$value
  .bnn_quad_positive(c(u, phi), "Local quadrature moments")
  entropy_cost <- .bnn_quad_integrate(function(z) {
    h <- base$offset(base$scale*z)
    ifelse(is.finite(h), -h*exp(h), 0)
  }, -Inf, Inf, rel_tol, subdivisions)
  entropy <- log(base$scale) + log(base$integral$value) + zeta +
    entropy_cost$value / base$integral$value
  list(log_normalizer = base$log_value, u = u, phi = phi, zeta = zeta,
       entropy = entropy, A = A, B = B, d = d,
       identity_residual = B * phi - A * u - (d - 1),
       estimated_relative_integration_error = max(
         base$integral$abs_error / base$integral$value,
         if (!is.null(first$integral)) first$integral$abs_error / first$integral$value else NA_real_,
         inverse$integral$abs_error / inverse$integral$value))
}

.bnn_quad_tail_moments <- function(K, C, mu, rho, rel_tol, subdivisions, use_score) {
  .bnn_quad_quadrature_settings(rel_tol, subdivisions)
  .bnn_quad_positive(c(K, rho), "Tail dimensions/reference scale")
  if (length(K) != 1L || K != as.integer(K) || length(C) != 1L ||
      !is.finite(C) || length(mu) != 1L || !is.finite(mu) || length(rho) != 1L)
    stop("Invalid tail kernel parameters.")
  boundary <- -log(2)
  log_kernel <- function(eta) stats::dnorm(eta, mu, rho, log = TRUE) +
    .bnn_quad_tail_g(eta, K, C)
  derivative <- function(eta) {
    r <- exp(-eta)
    gp <- r * (K * eta + C - K + K * digamma(1 + r) +
                 K * digamma(1 + K * r)) - K
    -(eta - mu) / rho^2 + gp
  }
  upper <- max(boundary + 8, mu + 12 * rho, log1p(abs(C) + K) + 8)
  for (iteration in 1:20) {
    grid <- seq(boundary, upper, length.out = 513L)
    heights <- log_kernel(grid)
    interior <- which(heights[2:512] >= heights[1:511] &
                       heights[2:512] >= heights[3:513]) + 1L
    candidates <- c(boundary, upper)
    if (length(interior)) for (j in interior) candidates <- c(candidates,
      stats::optimize(log_kernel, c(grid[j - 1L], grid[j + 1L]), maximum = TRUE,
               tol = 1e-11)$maximum)
    mode <- candidates[which.max(log_kernel(candidates))]
    peak <- log_kernel(mode)
    if (log_kernel(upper) < peak - 50 && derivative(upper) < 0) break
    upper <- upper + max(8 * rho, upper - boundary)
  }
  if (!(log_kernel(upper) < peak - 50 && derivative(upper) < 0))
    stop("Could not stabilize the tail quadrature range.")
  boundary_case <- mode == boundary
  if (boundary_case && derivative(boundary) > 0) {
    # A narrow maximum can lie between the truncation boundary and the very
    # first grid point. For the fitted N(0,1) log-shape prior the log kernel,
    # written as a function of r=1/xi, is strictly concave on 0 < r <= 2.
    # Find the first descending zero even if both sampled heights miss it.
    right <- grid[which(derivative(grid) <= 0)[1L]]
    mode <- stats::uniroot(derivative, c(boundary, right), tol = 1e-14)$root
    peak <- log_kernel(mode)
  }
  rmode <- exp(-mode)
  gp <- derivative(mode) + (mode - mu) / rho^2
  curvature <- 1 / rho^2 + gp + K - K * rmode +
    K * rmode^2 * trigamma(1 + rmode) + K^2 * rmode^2 * trigamma(1 + K * rmode)
  if (mode == boundary) {
    scale <- 1 / (abs(derivative(boundary)) + sqrt(max(curvature, 1/rho^2)))
  } else {
    scale <- if (is.finite(curvature) && curvature > 0) 1 / sqrt(curvature) else rho
  }
  .bnn_quad_positive(scale, "Tail quadrature scale")
  lower_z <- (boundary - mode) / scale
  other_modes <- (candidates - mode) / scale
  breaks <- as.vector(outer(other_modes, c(-1, 0, 1), "+"))
  if (boundary_case) {
    # Do not insert the distant grid endpoint as a quadrature breakpoint.
    # An adaptive rule on [8, many thousands] can miss the exponential
    # shoulder of a boundary maximum and falsely report a tiny error.
    breaks <- c(-32,-16,-8,-4,-2,-1,0,1,2,4,8,16,32,64)
  }
  offset <- function(eta) {
    value <- log_kernel(eta) - peak
    if (use_score) {
      # If large nearly equal log densities defeat both QUADPACK tolerances,
      # integrate the score over the displacement from the mode. Four
      # Gauss–Legendre nodes resolve the smoothly varying score over each
      # short displacement. Distant tails can use direct subtraction.
      delta <- eta - mode
      near <- abs(delta) < .15
      h <- delta[near]
      nodes <- c(.0694318442029737,.3300094782075719,.6699905217924281,.9305681557970262)
      weights <- c(.1739274225687269,.3260725774312731,.3260725774312731,.1739274225687269)
      average <- numeric(length(h))
      for (j in seq_along(nodes))
        average <- average + weights[j]*derivative(mode + h*nodes[j])
      value[near] <- h*average
    }
    value
  }
  density <- function(z) {
    exponent <- offset(mode + scale * z)
    exponent[!is.finite(exponent)] <- -Inf
    exp(exponent)
  }
  # Large layers can lose a few digits subtracting the log densities at eta
  # and at the mode. Retry only a panel whose requested tolerance failed.
  roundoff_tol <- max(rel_tol, 1e-8)
  integral <- .bnn_quad_integrate(density, lower_z, Inf,
    rel_tol, subdivisions, extra = breaks, roundoff_tol = roundoff_tol)
  .bnn_quad_positive(integral$value, "Tail quadrature integral")
  retried <- integral$retried
  expectation <- function(f) {
    ans <- .bnn_quad_integrate(function(z) {
      eta <- mode + scale * z
      w <- density(z)
      result <- numeric(length(z))
      positive <- w > 0
      result[positive] <- f(eta[positive]) * w[positive]
      result
    }, lower_z, Inf, rel_tol, subdivisions, extra = breaks, roundoff_tol = roundoff_tol)
    if (ans$retried) retried <<- TRUE
    ans$value / integral$value
  }
  logZ <- peak + log(scale) + log(integral$value)
  log_support <- stats::pnorm((mu - boundary) / rho, log.p = TRUE)
  r <- if (boundary_case) {
    2 * (1 + expectation(function(eta) expm1(boundary - eta)))
  } else {
    expectation(function(eta) exp(-eta))
  }
  normal_log <- expectation(function(eta) stats::dnorm(eta, mu, rho, log = TRUE))
  # A large peak makes the individual log density and entropy contributions
  # poorly conditioned even if the requested integration tolerance succeeded.
  centered_entropy <- use_score || boundary_case || abs(peak)*rel_tol > 1e-4
  if (!centered_entropy && !retried)
    g_mean <- expectation(function(eta) .bnn_quad_tail_g(eta, K, C))
  if (centered_entropy || retried) {
    expected_offset <- expectation(offset)
    g_mean <- peak + expected_offset - normal_log
    entropy <- log(scale) + log(integral$value) - expected_offset
  } else {
    entropy <- logZ - normal_log - g_mean
  }
  if (!is.finite(r) || r <= 0 || r > 2) stop("Tail quadrature violated 0 < E[1/xi] <= 2.")
  list(log_normalizer = logZ, r = r, eta_mean = expectation(identity),
       log_prior_support = log_support, expected_log_reference = normal_log,
       expected_log_factor = g_mean, entropy_eta = entropy,
       C = C, K = K, mu = mu, rho = rho, mode = mode,
       estimated_relative_integration_error = integral$abs_error / integral$value)
}

  }

gltbnn_shrinkage <- function(fit) {
  if (!inherits(fit, "gltbnn_fit")) {
    stop("fit must be a model returned by gltbnn_fit().", call. = FALSE)
  }
  state <- fit$engine$state
  log_reference_variance <- if (is.null(state$noise)) 0 else state$noise$zeta
  coefficient_geometry <- identical(fit$engine$config$geometry, "coefficient")
  by_layer <- lapply(seq_along(fit$hidden), function(layer) {
    width <- fit$hidden[layer]
    dimension <- fit$engine$config$d[layer]
    group <- state$g[[layer]]
    if (coefficient_geometry) {
      unit <- rep(seq_len(width), each = dimension)
      coordinate <- rep(seq_len(dimension), width)
      # Scale factors follow the weight matrix's column-major storage order.
      index <- unit + (coordinate - 1L) * width
      rows <- data.frame(layer = layer, unit = unit, coordinate = coordinate,
                         is_bias = coordinate == dimension, group_size = 1L)
    } else {
      index <- seq_len(width)
      rows <- data.frame(layer = layer, unit = index, group_size = dimension)
    }
    terms <- .bnn_prior_terms(group, length(index), fit$prior)
    # The inverse-gamma factor represents tau under GLT and tau^2 under G/GL.
    # For GLT, tail$r = E_q[1/xi] and tail$eta_mean = E_q[log xi].
    log_layer_scale <- if (fit$prior == "GLT") group$tau$zeta else group$tau$zeta / 2
    rows$scale_precision <- terms$precision[index]
    rows$geometric_prior_sd <- exp((log_reference_variance + terms$logvar[index]) / 2)
    rows$geometric_layer_scale <- exp(log_layer_scale)
    rows$geometric_tail_shape <- if (fit$prior == "GLT") exp(group$tail$eta_mean) else NA_real_
    rows
  })
  do.call(rbind, by_layer)
}
